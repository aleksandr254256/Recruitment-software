  <div class="tab-panel" id="tab-help">
    <section class="block">
      <div class="head"><h2>Как пользоваться инструментом</h2></div>
      <div style="padding:18px 20px; font-size:13.5px; line-height:1.7;">
        <h3 style="color:var(--blueprint); margin-top:0;">1. Загрузка данных</h3>
        <p>Загрузите два CSV-файла: «Сбор данных» (обязателен — это основной лог кандидатов) и «Отчет» (нужен только для списка открытых заявок). Файлы загружаются вручную каждый раз, когда нужен свежий отчёт — данные не обновляются сами.</p>

        <h3 style="color:var(--blueprint);">2. Фильтры сверху</h3>
        <p>Две группы: <b>Период</b> (День/Неделя/Месяц + Год) и <b>срез данных</b> (ОП, Проект, Вид/участок, Должность, Категория). <b>Рекрутер</b> — отдельно, справа.<br>
        <b>Год</b> — год события, считается от даты статуса кандидата.<br>
        <b>ОП</b> — площадка. При выборе ОП список проектов сужается только до проектов этой площадки.<br>
        <b>Проект</b> — конкретный проект в рамках выбранного ОП.<br>
        <b>Должность</b> — поиск по вхождению текста (например, «монтажник» найдёт все вариации должности).<br>
        <b>Вид / участок</b> — производственный участок (МК, МО, РВС, ТТ и т.д.).<br>
        <b>Категория</b> — ИТР / ИТР-Лин. / РОП / РОП-Сварщик.<br>
        <b>Рекрутер</b> — фильтр по конкретному рекрутеру, который ведёт кандидата.<br>
        <b>Период</b> — один общий переключатель на весь инструмент: День / Неделя / Месяц / **Все**. Он управляет сразу всеми показателями, которые зависят от времени: новыми откликами, активностью, «Направлено» и отказами. «Все» — без ограничения по дате, считает за всё время. Неделя по умолчанию — текущая календарная (от сегодняшней даты), считается с понедельника по воскресенье и ограничена неделями выбранного года. Готовность к выезду (Оперативный отчёт) в этот переключатель не входит — у неё свой фиксированный горизонт, ближайший месяц от сегодня.</p>

        <h3 style="color:var(--blueprint);">3. Вкладки</h3>
        <p><b>Дашборд</b> (открывается первой) — снэпшот на сегодня, не зависит от «Период», порядок сверху вниз: два кольца — «Открыто заявок из N» и «Осталось закрыть вакансий из N» — оба следуют за фильтром «Год» сверху: если выбран конкретный год, считают заявки с датой поступления в этом году; если выбрано «Все», считают по всем годам разом (сейчас это 2025–2026), подпись под кольцами показывает выбранный период. В центре первого кольца — сколько из этих заявок сейчас в статусе «Открыта»; в центре второго — сумма «Кол-во вакансий» минус Куплен билет/на ОП/Трудоустроен по кандидатам, привязанным к ЭТИМ ЖЕ, но только открытым заявкам (закрытые в остаток не считаются). Дальше — баннер «Динамика на этой неделе»; столбчатая диаграмма «Потребность по ОП» (закрыли / осталось); таблица «Потребность по заявкам» (строки — категория, столбцы — ОП); «Потребность по категориям»; и в самом низу «Этапы воронки» — 13 статусов плитками (в шапке блока справа — «В работе»; под числом каждой плитки — «+N за нед.», сколько кандидатов вошло в этот статус на текущей календарной неделе).<br>
        В «Этапах воронки» «Через ЦТС» (канал, только активные кандидаты) стоит сразу после «Мобилизации» — отдельно от статуса «на ЦТС», это разные вещи: один канал маршрутизации, другой — текущий статус.<br>
        <b>Сводка за период</b> — три карточки сверху: новые отклики («сколько нашли»), активность (движение по статусам) и «Направлено» (кто получил статус «на ОП» / «Трудоустроен» / «Куплен билет» за период). Под карточками — сворачиваемые блоки, по умолчанию свёрнуты, кликните на заголовок блока, чтобы развернуть: «Закрыли за неделю» — кого фактически отправили за период, с разбивкой «куда» (по ОП/проектам) и «кого» (по должностям), плюс полный список кандидатов; список новых откликов; и разбивка активности по статусам. Готовности к выезду здесь больше нет — она переехала на «Оперативный отчёт», чтобы не было двух разных ответов на один вопрос «когда приедут».<br>
        <b>Оперативный отчёт</b> — открытые заявки (с итоговой строкой по сумме вакансий/осталось — строка закреплена внизу таблицы), кто сейчас «в работе», диаграмма «в работе по ОП» и сводная таблица готовности к выезду: недели сгруппированы сверху, даты — под ними. Эта сводная таблица всегда показывает фиксированный горизонт — ближайший месяц от сегодняшнего дня, а не выбранный период. Клик по любой ячейке с числом мгновенно показывает список конкретных кандидатов (включая гражданство и источник).<br>
        <b>Отказы</b> — сколько откликнулось за период и сколько из них отказано в течение этого же периода, с разбивкой по причинам и диаграммой.<br>
        <b>Заявки на контроле</b> — заявки, которые открыты 60+ дней без закрытия (не зависит от периода — это снэпшот на сегодня); ниже — открытые заявки, у которых нет ни одного привязанного кандидата в файле «Сбор данных» (обычно из-за несовпадения номера заявки в поле «Заявка» с «№ Заявки» в «Отчете») — эти заявки стоит сверить в исходниках, т.к. по ним «Осталось закрыть» считается без учёта реального прогресса.<br>
        <b>Рекрутеры</b> — нагрузка и показатели по каждому рекрутеру: сколько кандидатов «в работе» сейчас, «Доходимость» — сколько дошло до решённого исхода (Куплен билет/на ОП/Трудоустроен/Отказ), «% Доходимости» — какая доля из дошедших до решения оказалась успешной, а не отказом (считается именно от дошедших, не от всех когда-либо назначенных — иначе рекрутер со свежим набором кандидатов выглядел бы хуже просто из-за возраста своей воронки), среднее число дней без смены статуса у активных кандидатов, и отдельно — счётчик дат статуса, ошибочно попавших в будущее. Ниже — эффективность передачи кандидатов между рекрутерами: «Всего получено/передано» — сколько кандидатов прошло через передачу, «Из них трудоустроены» / «Из них отказались» / «Ещё в работе» — их исход, «% трудоустройства» — доля успешных среди уже решённых. Связка «от кого — кому» в исходных данных не хранится напрямую, восстановлена по комментарию/ФИО+ОП+Проекту/телефону, точный процент покрытия виден прямо в блоке. Уволенные рекрутеры (Тхорева Е, Попов Д, Голубкина С) исключены из всего трекера целиком. Не зависит от «Период» — это накопленная картина по всем кандидатам рекрутера.</p>

        <h3 style="color:var(--blueprint);">4. Сортировка, печать и экспорт</h3>
        <p>В таблицах «Заявки», «В работе» и «Заявки на контроле» можно кликнуть на заголовок столбца, чтобы отсортировать по нему — повторный клик меняет направление. Кнопка «Печать / PDF» в шапке страницы открывает системное окно печати браузера — там можно сохранить текущий открытый экран в PDF. У таблиц «Заявки (открытые)» и «В работе» есть отдельная кнопка «Экспорт в Excel» — она выгружает именно то, что сейчас отфильтровано и отсортировано на экране, в .xlsx-файл.</p>

        <h3 style="color:var(--blueprint);">5. Важно понимать про логику дат</h3>
        <p>Номер недели (КН) программа считает <b>сама</b>, по дате события — не берёт готовое число из исходной таблицы. Неделя идёт с понедельника по воскресенье. Если дата в исходнике набрана с ошибкой (например, неверный год или обрезанная запись), неделя посчитается неверно. Проверяйте подозрительно большие/маленькие номера недель на предмет опечаток в датах.</p>

  if(y.length===2) y = "20"+y;
  const yearNum = parseInt(y);
  if(y.length!==4 || yearNum<2000 || yearNum>2100) return null; // неправдоподобный год (напр. обрыв "...202" вместо "...2026") — считаем дату нераспознанной
  const dt = new Date(yearNum, parseInt(mo)-1, parseInt(d));
  if(isNaN(dt.getTime())) return null;
  return dt;
}
// КН считаем сами по дате: неделя с понедельника по воскресенье (стандарт ISO 8601, по вашему запросу).
function isoWeekParts(date){
  const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
  const dayNum = d.getUTCDay() || 7; // Пн=1 ... Вс=7
  d.setUTCDate(d.getUTCDate() + 4 - dayNum);
  const yearStart = new Date(Date.UTC(d.getUTCFullYear(),0,1));
  const week = Math.ceil((((d - yearStart) / 86400000) + 1)/7);
  return {year: d.getUTCFullYear(), week: week};
}
function computeKN(dateStr){
  const d = parseRUDate(dateStr);
  if(!d) return {year:null, week:null};
  return isoWeekParts(d); // неделя Пн-Вс, стандарт ISO 8601
}
// календарный месяц (для отчёта по отказам за месяц) — обычный месяц/год, без сдвигов недель
function computeMonth(dateStr){
  const d = parseRUDate(dateStr);
  if(!d) return {year:null, month:null};
  return {year: d.getFullYear(), month: d.getMonth()+1};
}

// Восстановление ОП/Проекта/Участка у кандидата по номеру заявки, если поле "Заявка" совпадает
// с "№ Заявки" в файле "Отчет", а сами поля у кандидата пустые. Не трогает уже заполненные поля.
function backfillFromReport(){
  if(!collectionData.length || !reportData.length) return 0;
  const zayavkaLookup = new Map();
  reportData.forEach(r=>{ if(r.zayavkaNum && !zayavkaLookup.has(r.zayavkaNum)) zayavkaLookup.set(r.zayavkaNum, r); });
  let filled = 0;
  collectionData.forEach(r=>{
    if(!r.zayavka || /^без заявки$/i.test(r.zayavka)) return;
    if(r.op && r.project && r.uchastok) return; // всё уже заполнено — восстанавливать нечего
    const match = zayavkaLookup.get(r.zayavka);
    if(!match) return;
    let touched = false;
    if(!r.op && match.op){ r.op = match.op; touched = true; }
    if(!r.project && match.project){ r.project = match.project; touched = true; }
    if(!r.uchastok && match.uchastok){
      r.uchastok = canonUchastok(match.uchastok);
      r.vid = getVid(r.uchastok);
      touched = true;
    }
    if(touched) filled++;
  });
  return filled;
}

// columns: [{label, value(row)}, ...]; экспортирует именно то, что сейчас отфильтровано/отсортировано и видно в таблице
function exportRowsToExcel(rows, columns, filenameBase){
  if(!rows || !rows.length){ alert("Нет данных для экспорта по текущему фильтру."); return; }
  const data = [columns.map(c=>c.label)].concat(rows.map(r=>columns.map(c=>c.value(r)??"")));
  const ws = XLSX.utils.aoa_to_sheet(data);
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, "Данные");
  const stamp = new Date().toISOString().slice(0,10);
  XLSX.writeFile(wb, `${filenameBase}_${stamp}.xlsx`);
}

function populateFilterOptions(){
  const years = [...new Set(collectionData.map(r=>r.year).filter(y=>y!==null && y>=2000 && y<2050))].sort((a,b)=>a-b);
  const ops = [...new Set(collectionData.map(r=>r.op).filter(Boolean))].sort();
  const categories = [...new Set(collectionData.map(r=>r.category).filter(Boolean))].sort();
  const recruiters = [...new Set(collectionData.map(r=>r.recruiter).filter(Boolean))].sort();
  const vids = [...new Set(collectionData.map(r=>r.vid).filter(Boolean))].sort();

  fillSelect("f-year", years);
  fillSelect("f-op", ops);
  fillSelect("f-vid", vids);
  fillSelect("f-category", categories);
  fillSelect("f-recruiter", recruiters);

  // по умолчанию выбираем 2026 год, если он есть в данных
  if(years.includes(2026)){
    document.getElementById("f-year").value = 2026;
  }
  refreshProjectOptions();
  refreshPeriodOptions();
}

  fillSelect("f-project", projects);
}

function matchesBaseFilters(r, f){
  if(f.op && r.op!==f.op) return false;
  if(f.vid && r.vid!==f.vid) return false;
  if(f.project && r.project!==f.project) return false;
  if(f.position && !normKey(r.position).includes(f.position)) return false;
  if(f.category && r.category && r.category!==f.category) return false;
  if(f.recruiter && r.recruiter && r.recruiter!==f.recruiter) return false;
  return true;
}

const sortState = {}; // tableId -> {key, dir}
function sortableHeader(tableId, label, key, extraClass){
  const state = sortState[tableId];
  const active = state && state.key===key;
  const arrow = active ? (state.dir===1 ? " ▲" : " ▼") : "";
  const cls = extraClass ? ("sortable " + extraClass) : "sortable";
  return `<th class="${cls}" data-table="${tableId}" data-key="${esc(key)}">${esc(label)}${arrow}</th>`;
}
function applySorting(tableId, rows, accessors){
  const state = sortState[tableId];
  if(!state || !accessors[state.key]) return rows;
  const acc = accessors[state.key];
  return [...rows].sort((a,b)=>{
    let av = acc(a), bv = acc(b);
    if(av===null||av===undefined) av = "";
    if(bv===null||bv===undefined) bv = "";
    if(typeof av==="string") av = av.toLowerCase();
    if(typeof bv==="string") bv = bv.toLowerCase();
    if(av<bv) return -1*state.dir;
    if(av>bv) return 1*state.dir;
    return 0;
  });
}
document.addEventListener("click", function(e){
  const th = e.target.closest("th.sortable");
  if(!th) return;
  const tableId = th.dataset.table, key = th.dataset.key;
  const cur = sortState[tableId];
  if(cur && cur.key===key){ cur.dir *= -1; } else { sortState[tableId] = {key, dir:1}; }
  renderAll();
});

function renderStackedColumnChart(el, entries){
  if(!entries.length){ el.innerHTML = '<div class="empty-state">Нет данных для диаграммы.</div>'; return; }
  const w = Math.max(500, entries.length*90), h = 310, padL = 40, padR = 20, padT = 46, padB = 60;
  const plotW = w - padL - padR, plotH = h - padT - padB;
  const maxV = Math.max(1, ...entries.map(e=>e.total));
  const barGap = 14;
  const barW = (plotW/entries.length) - barGap;
  const closedColor = "var(--blueprint)", remainingColor = "var(--silver)";
  const MIN_LABEL_SEG = 16; // сегмент выше этой высоты — подпись по центру сегмента (белым); ниже — подпись переносится наверх, рядом с итогом
  const yBase = padT + plotH;
  // сжимающая шкала (корень) — уменьшает разрыв между самым большим и самым маленьким столбцом,
  // при этом порядок величин сохраняется и доля "закрыли/осталось" внутри каждого столбца считается от его
  // РЕАЛЬНОГО (несжатого) итога, чтобы сама заливка сегментов не искажалась
  // каждый ненулевой столбец получает не меньше 22% высоты графика — иначе маленькие ОП (3-14 вакансий)
  // рядом с Усть-Лугой (1258) превращаются в полоску толщиной в пиксель. Дальше — сжатая (корень) шкала.
  const MIN_BAR_FRAC = 0.22;
  const scaleH = (v)=> v>0 ? plotH * (MIN_BAR_FRAC + (1-MIN_BAR_FRAC) * Math.sqrt(v)/Math.sqrt(maxV)) : 0;
  let svg = `<svg viewBox="0 0 ${w} ${h}" style="width:100%; height:auto; font-family:var(--font-ui);">`;
  entries.forEach((e,i)=>{
    const x = padL + i*(plotW/entries.length) + barGap/2;
    const cx = x + barW/2; // центр столбца — все подписи выравниваем по нему
    const totalH = scaleH(e.total);
    let closedH = e.total ? (e.closed/e.total)*totalH : 0;
    // если синий сегмент слишком тонкий для своей же подписи — раздвигаем его до минимума за счёт серого,
    // чтобы цифра "закрыли" всегда была внутри синей заливки, а не улетала наверх столбца
    if(e.closed>0 && closedH < MIN_LABEL_SEG) closedH = Math.min(totalH, MIN_LABEL_SEG);
    const remainH = totalH - closedH;
    const yTop = padT + plotH - totalH;
    const yClosedTop = padT + plotH - closedH;
    const remaining = e.total - e.closed;

    // верхний сегмент (осталось)
    if(remainH>0.5) svg += `<rect x="${x}" y="${yTop}" width="${barW}" height="${remainH}" fill="${remainingColor}" rx="2"/>`;
    // нижний сегмент (закрыли)
    if(closedH>0.5) svg += `<rect x="${x}" y="${yClosedTop}" width="${barW}" height="${closedH}" fill="${closedColor}" rx="2"/>`;

    // "осталось" — по центру своего сегмента, если он достаточно высокий; "закрыли" — теперь всегда внутри (см. выше)
    if(remaining>0 && remainH >= MIN_LABEL_SEG){
      svg += `<text x="${cx}" y="${yTop+remainH/2+4}" font-size="10" text-anchor="middle" fill="var(--ink)" font-family="var(--font-mono)" font-weight="700">${remaining}</text>`;
    }
    if(e.closed>0){
      svg += `<text x="${cx}" y="${yClosedTop+closedH/2+4}" font-size="10" text-anchor="middle" fill="#fff" font-family="var(--font-mono)" font-weight="700">${e.closed}</text>`;
    }

    // слишком тонкий серый сегмент — подпись не помещается внутри, переносим наверх столбца,
    // а не под ось, чтобы не наезжала на повёрнутые подписи ОП снизу
    const stacked = [];
    if(remaining>0 && remainH < MIN_LABEL_SEG) stacked.push({text:remaining, color:"var(--ink-soft)"});
    let curY = yTop - 10;
    stacked.forEach(lbl=>{
      svg += `<text x="${cx}" y="${curY}" font-size="10" text-anchor="middle" fill="${lbl.color}" font-family="var(--font-mono)" font-weight="700">${lbl.text}</text>`;
      curY -= 12;
    });
    // итого — всегда самая верхняя подпись над столбцом, по центру
    svg += `<text x="${cx}" y="${curY}" font-size="11" text-anchor="middle" fill="var(--ink)" font-family="var(--font-mono)" font-weight="700">${e.total}</text>`;

    svg += `<text x="${cx}" y="${h-36}" font-size="10.5" text-anchor="middle" fill="var(--ink-soft)" transform="rotate(-20 ${cx} ${h-36})">${esc(e.label)}</text>`;
  });
  svg += `<line x1="${padL}" y1="${yBase}" x2="${w-padR}" y2="${yBase}" stroke="var(--line-strong)" stroke-width="1"/>`;
  svg += `</svg>`;
  svg += `<div style="display:flex; gap:16px; margin-top:8px; font-size:11.5px; color:var(--ink-soft);">
    <span><span style="display:inline-block; width:10px; height:10px; background:var(--blueprint); border-radius:2px; margin-right:5px;"></span>Закрыли (Трудоустроен/Перевод/на ОП/Куплен билет)</span>
    <span><span style="display:inline-block; width:10px; height:10px; background:var(--silver); border-radius:2px; margin-right:5px;"></span>Осталось</span>
  </div>`;
  el.innerHTML = svg;
}

function renderPivot(f){
  const rows = collectionData.filter(r=> ACTIVE_STATUSES.includes(r.status) && matchesBaseFilters(r, f) && (!f.year || r.knStatusYear===f.year));
  const today = new Date(); today.setHours(0,0,0,0);
  const horizon = new Date(today.getTime() + 31*24*60*60*1000);
  const withDate = rows.filter(r=>{
    if(!r.readyDate) return false;
    const d = parseRUDate(r.readyDate);
    return d && d>=today && d<=horizon;
  });
  const withoutDate = rows.filter(r=>!r.readyDate || !parseRUDate(r.readyDate));

  document.getElementById("cnt-pivot").textContent = (withDate.length+withoutDate.length) + " чел.";
  const tbl = document.getElementById("tbl-pivot");
  const visibleRows = withDate.concat(withoutDate); // именно то, что реально попадёт в таблицу
  if(!visibleRows.length){ tbl.innerHTML = '<tr><td class="empty-state">Нет данных для сводной таблицы в пределах горизонта планирования (1 месяц).</td></tr>'; return; }

  const positions = [...new Set(visibleRows.map(r=>r.position))].sort();

  // группируем даты по неделям: weekKey ("2026-W34") -> Set дат ("DD.MM.YYYY"), с сохранением сортировки
  const weekToDates = new Map(); // weekKey -> Map(dateKey -> timestamp) для сортировки
  withDate.forEach(r=>{
    const d = parseRUDate(r.readyDate);
    const kn = isoWeekParts(d);
    const weekKey = kn.year + "-W" + String(kn.week).padStart(2,"0");
    const dateKey = String(d.getDate()).padStart(2,"0")+"."+String(d.getMonth()+1).padStart(2,"0")+"."+d.getFullYear();
    if(!weekToDates.has(weekKey)) weekToDates.set(weekKey, new Map());
    weekToDates.get(weekKey).set(dateKey, d.getTime());
  });
  const weekKeysSorted = [...weekToDates.keys()].sort();
  const hasNoDate = withoutDate.length>0;

  // группируем кандидатов по ячейке (должность+дата) — для drill-down по клику
  const cellCandidates = {};
  rows.forEach(r=>{
    let key;
    if(r.readyDate && parseRUDate(r.readyDate)){
      const d = parseRUDate(r.readyDate);
      key = String(d.getDate()).padStart(2,"0")+"."+String(d.getMonth()+1).padStart(2,"0")+"."+d.getFullYear();
    } else { key = "—"; }
    const cellKey = r.position+"|||"+key;
    (cellCandidates[cellKey] = cellCandidates[cellKey] || []).push(r);
  });
  window.__pivotCells = cellCandidates;

  // строим двухуровневый заголовок: КН недели (colspan) сверху, конкретные даты + "Итого КН" снизу
  let headRow1 = '<tr><th class="rowlabel" rowspan="2">Должность</th>';
  let headRow2 = '<tr>';
  const weekGroups = []; // [{weekKey, dates:[...]}]
  const weekStartDates = new Set(); // первая дата каждой недели — для визуальной границы
  weekKeysSorted.forEach(weekKey=>{
    const dateMap = weekToDates.get(weekKey);
    const dates = [...dateMap.keys()].sort((a,b)=>dateMap.get(a)-dateMap.get(b));
    if(dates.length) weekStartDates.add(dates[0]);
    weekGroups.push({weekKey, dates});
    // colspan = кол-во дат + 1 (промежуточный итог входит в ту же КН-группу визуально)
    headRow1 += `<th colspan="${dates.length+1}" style="text-align:center; border-left:2px solid var(--line-strong);">${weekKey.replace("-W"," · КН ")}</th>`;
    dates.forEach((dk,i)=>{
      headRow2 += `<th style="${i===0?'border-left:2px solid var(--line-strong);':''}">${dk.slice(0,5)}</th>`;
    });
    headRow2 += `<th style="background:#FAFAF7; color:var(--ink); font-weight:700;">Итого КН</th>`;
  });
  if(hasNoDate){
    headRow1 += '<th rowspan="2" style="border-left:2px solid var(--line-strong);">Без даты</th>';
  }
  headRow1 += '<th rowspan="2">Итого</th></tr>';
  headRow2 += '</tr>';

  let html = '<thead>' + headRow1 + headRow2 + '</thead><tbody>';
  positions.forEach(p=>{
    let total = 0;
    let cells = "";
    weekGroups.forEach(group=>{
      let weekSubtotal = 0;
      group.dates.forEach((dk,i)=>{
        const key = p+"|||"+dk;
        const list = cellCandidates[key] || [];
        weekSubtotal += list.length;
        const borderStyle = weekStartDates.has(dk) ? 'border-left:2px solid var(--line-strong);' : '';
        const cls = list.length
          ? `style="cursor:pointer; ${borderStyle}" class="pivot-cell" data-key="${esc(key)}"`
          : (borderStyle ? `style="${borderStyle}"` : "");
        cells += `<td ${cls}>${list.length||""}</td>`;
      });
      total += weekSubtotal;
      cells += `<td style="background:#FAFAF7; font-weight:700;">${weekSubtotal||""}</td>`;
    });
    if(hasNoDate){
      const key = p+"|||—";
      const list = cellCandidates[key] || [];
      total += list.length;
      const cls = list.length ? 'style="cursor:pointer; border-left:2px solid var(--line-strong);" class="pivot-cell" data-key="'+esc(key)+'"' : 'style="border-left:2px solid var(--line-strong);"';
      cells += `<td ${cls}>${list.length||""}</td>`;
    }
    html += `<tr><td class="rowlabel">${esc(p)}</td>${cells}<td><b>${total}</b></td></tr>`;
  });
  html += "</tbody>";
  tbl.innerHTML = html;

  tbl.querySelectorAll(".pivot-cell").forEach(td=>{
    td.addEventListener("click", function(){
      const key = td.dataset.key;
      const list = window.__pivotCells[key] || [];
      const [pos, dateKey] = key.split("|||");
      document.getElementById("pivot-modal-title").textContent = `${pos} — ${dateKey==="—"?"без даты готовности":dateKey} (${list.length} чел.)`;
      document.getElementById("pivot-modal-table").innerHTML =
        '<thead><tr><th>ФИО</th><th>ОП</th><th>Проект</th><th>Гражданство</th><th>Источник</th><th>Статус</th><th>Дата готовности</th><th>Рекрутер</th></tr></thead><tbody>' +
        list.map(r=>`<tr><td>${esc(r.fio)}</td><td>${esc(r.op)}</td><td>${esc(r.project)}</td><td>${esc(r.citizenship)}</td><td>${esc(r.source)}</td><td><span class="badge ${statusBadgeClass(r.status)}">${esc(r.status)}</span></td><td>${esc(r.readyDate)}</td><td>${esc(r.recruiter)}</td></tr>`).join("") +
        "</tbody>";
      document.getElementById("pivot-modal").style.display = "flex";

// Универсальная проверка попадания даты кандидата в выбранный период (день/неделя/месяц)
function matchesPeriod(r, kind){
  // kind: 'response' -> дата отклика, 'status' -> дата статуса, 'ready' -> дата готовности
  const rawDateField = {response:"responseDate", status:"statusDate", ready:"readyDate"}[kind];
  const monthField = {response:"respMonth", status:"statMonth", ready:"readyMonth"}[kind];
  const monthYearField = {response:"respMonthYear", status:"statMonthYear", ready:"readyMonthYear"}[kind];
  const weekField = {response:"knResponse", status:"knStatus", ready:"knReady"}[kind];
  const weekYearField = {response:"knResponseYear", status:"knStatusYear", ready:"knReadyYear"}[kind];

  const type = document.getElementById("period-type").value;
  if(type==="all") return true; // без ограничения по дате — считаем всех, вне зависимости от периода
  if(type==="day"){
    const target = document.getElementById("period-day").value; // YYYY-MM-DD
    if(!target) return false;
    const d = parseRUDate(r[rawDateField]);
    if(!d) return false;
    const key = d.getFullYear()+"-"+String(d.getMonth()+1).padStart(2,"0")+"-"+String(d.getDate()).padStart(2,"0");
    return key===target;
  }
  if(type==="month"){
    const val = document.getElementById("period-month").value; // "YYYY-M"
    if(!val) return false;
    const [y,m] = val.split("-").map(Number);
    return r[monthYearField]===y && r[monthField]===m;
  }
  // неделя (по умолчанию)
  const val = document.getElementById("period-week").value; // "YYYY-W"
  if(!val) return false;
  const [y,w] = val.split("-W").map(Number);
  const py = r[weekYearField];
  const pw = r[weekField];
  return py===y && pw===w;
}

  if(type==="month") return !!document.getElementById("period-month").value;
  return !!document.getElementById("period-week").value;
}
function refreshPeriodOptions(){
  const yearVal = document.getElementById("f-year").value;
  const year = yearVal ? parseFloat(yearVal) : null;
  // недели и месяцы считаем только внутри выбранного глобального года — иначе редкие ошибочные годы (напр. 2066) перетягивают дефолт
  const weekSet = new Set();
  const monthSet = new Set();
  collectionData.forEach(r=>{
    if(r.knStatus!==null && r.knStatus<=53 && (year===null || r.knStatusYear===year)) weekSet.add(r.knStatusYear+"-W"+String(r.knStatus).padStart(2,"0"));
    if(r.knResponse!==null && r.knResponse<=53 && (year===null || r.knResponseYear===year)) weekSet.add(r.knResponseYear+"-W"+String(r.knResponse).padStart(2,"0"));
    if(r.knReady!==null && r.knReady<=53 && (year===null || r.knReadyYear===year)) weekSet.add(r.knReadyYear+"-W"+String(r.knReady).padStart(2,"0"));
    if(r.statMonth!==null && (year===null || r.statMonthYear===year)) monthSet.add(r.statMonthYear+"-"+r.statMonth);
    if(r.respMonth!==null && (year===null || r.respMonthYear===year)) monthSet.add(r.respMonthYear+"-"+r.respMonth);
    if(r.readyMonth!==null && (year===null || r.readyMonthYear===year)) monthSet.add(r.readyMonthYear+"-"+r.readyMonth);
  });
  const weeks = [...weekSet].sort();
  // по умолчанию — ТЕКУЩАЯ календарная неделя (от сегодняшней даты), а не максимум из данных
  const todayKN = isoWeekParts(new Date());
  const todayWeekKey = todayKN.year+"-W"+String(todayKN.week).padStart(2,"0");
  if((year===null || todayKN.year===year) && !weeks.includes(todayWeekKey)) weeks.push(todayWeekKey);
  weeks.sort();
  const months = [...monthSet].sort((a,b)=>{
    const [ay,am]=a.split("-").map(Number), [by,bm]=b.split("-").map(Number);
    return ay-by || am-bm;
  });
  const weekSel = document.getElementById("period-week");
  weekSel.innerHTML = weeks.map(w=>`<option value="${w}">${w.replace("-W"," · КН ")}</option>`).join("");
  if(year===null || todayKN.year===year) weekSel.value = todayWeekKey;
  else if(weeks.length) weekSel.value = weeks[weeks.length-1];
  const monthSel = document.getElementById("period-month");
  monthSel.innerHTML = months.map(m=>{ const [y,mm]=m.split("-").map(Number); return `<option value="${m}">${MONTH_NAMES[mm]} ${y}</option>`; }).join("");
  if(months.length) monthSel.value = months[months.length-1];

  // День: по умолчанию берём последнюю дату отклика в данных (в пределах выбранного года, если он задан)
  let maxRespDate = null;
  collectionData.forEach(r=>{
    if(year!==null && r.respMonthYear!==year) return;
    const d = parseRUDate(r.responseDate);
    if(d && (!maxRespDate || d>maxRespDate)) maxRespDate = d;
  });
  if(maxRespDate){
    const iso = maxRespDate.getFullYear()+"-"+String(maxRespDate.getMonth()+1).padStart(2,"0")+"-"+String(maxRespDate.getDate()).padStart(2,"0");
    document.getElementById("period-day").value = iso;
  }
}

// Данные не хранят "кому передали" напрямую. Восстанавливаем по приоритету методов:
// 1) комментарий отправителя называет рекрутера + есть его строка-результат по тому же ФИО
// 2) совпадение ФИО + ОП + Проект + более поздняя дата + другой рекрутер (ровно 1 кандидат)
// 3) совпадение телефона + более поздняя дата + другой рекрутер (ровно 1 кандидат)
// Покрытие на реальных данных — around 77% передач (точная цифра всегда актуальна прямо в интерфейсе, см. блок "Эффективность передачи"). Остальное честно помечается как "не удалось установить".
let transferResolutions = [];
function escapeRegex(s){ return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&"); }
function findRecruitersInComment(comment, knownRecruiters, exclude){
  if(!comment) return [];
  return knownRecruiters.filter(rec => rec!==exclude && new RegExp(escapeRegex(rec)+"\\.?(?!\\w)").test(comment));
}
function resolveTransfers(){
  const recruiterList = [...new Set(collectionData.map(r=>r.recruiter).filter(Boolean))];
  const byFio = new Map();
  collectionData.forEach(r=>{
    if(!byFio.has(r.fio)) byFio.set(r.fio, []);
    byFio.get(r.fio).push(r);
  });
  const byPhone = new Map();
  collectionData.forEach(r=>{
    if(r.phone){
      if(!byPhone.has(r.phone)) byPhone.set(r.phone, []);
      byPhone.get(r.phone).push(r);
    }
  });

  const transfers = collectionData.filter(r=>r.status==="Передан др. рекрутеру");
  const resolved = [];
  transfers.forEach(sendRow=>{
    const senderDt = parseRUDate(sendRow.statusDate);
    if(!senderDt) return;
    let receivingRow = null, method = null;

    // Метод 1: комментарий
    const mentioned = findRecruitersInComment(sendRow.comment, recruiterList, sendRow.recruiter);
    if(mentioned.length===1){
      const target = mentioned[0];
      const candidates = (byFio.get(sendRow.fio)||[]).filter(r2=>{
        const d = parseRUDate(r2.statusDate);
        return r2.recruiter===target && d && d>senderDt;
      });
      if(candidates.length>=1){
        candidates.sort((a,b)=> parseRUDate(a.statusDate) - parseRUDate(b.statusDate));
        receivingRow = candidates[0]; method = "comment";
      }
    }
    // Метод 2: ФИО+ОП+Проект+дата
    if(!receivingRow){
      const candidates = (byFio.get(sendRow.fio)||[]).filter(r2=>{
        const d = parseRUDate(r2.statusDate);
        return r2.op===sendRow.op && r2.project===sendRow.project && r2.recruiter!==sendRow.recruiter && d && d>senderDt;
      });
      if(candidates.length===1){ receivingRow = candidates[0]; method = "fio_op_proj"; }
    }
    // Метод 3: телефон
    if(!receivingRow && sendRow.phone){
      const candidates = (byPhone.get(sendRow.phone)||[]).filter(r2=>{
        const d = parseRUDate(r2.statusDate);
        return r2.recruiter!==sendRow.recruiter && d && d>senderDt;
      });
      const uniqRecruiters = [...new Set(candidates.map(r2=>r2.recruiter))];
      if(uniqRecruiters.length===1){
        candidates.sort((a,b)=> parseRUDate(a.statusDate) - parseRUDate(b.statusDate));
        receivingRow = candidates[0]; method = "phone";
      }
    }

    if(receivingRow){
      resolved.push({from: sendRow.recruiter, to: receivingRow.recruiter, method, sendRow, receivingRow});
    }
  });
  return resolved;
}

function renderRecruiters(f){
  const tbl = document.getElementById("tbl-recruiters");
  // берём всех кандидатов с рекрутером и статусом, без учёта самого фильтра "Рекрутер" (иначе таблица схлопнется в одну строку)
  const fNoRecruiter = {...f, recruiter: ""};
  const rows = collectionData.filter(r=> r.recruiter && r.status && matchesBaseFilters(r, fNoRecruiter) && (!f.year || r.knStatusYear===f.year));
  document.getElementById("cnt-recruiters").textContent = "";
  if(!rows.length){ tbl.innerHTML = '<tr><td class="empty-state">Нет данных по рекрутерам по текущему фильтру.</td></tr>'; return; }

  const today = new Date(); today.setHours(0,0,0,0);
  const byRecruiter = {};
  rows.forEach(r=>{
    const rec = r.recruiter;
    if(!byRecruiter[rec]) byRecruiter[rec] = {total:0, active:0, success:0, refused:0, activeDaysSum:0, activeDaysCount:0, dateAnomalies:0};
    const g = byRecruiter[rec];
    g.total++;
    if(ACTIVE_STATUSES.includes(r.status)) g.active++;
    if(STATUS_GROUP_2.includes(r.status)) g.success++; // дошёл до финала: Куплен билет / Трудоустроен / на ОП
    if(r.status==="Отказ") g.refused++;
    if(ACTIVE_STATUSES.includes(r.status)){
      const d = parseRUDate(r.statusDate);
      if(d){
        const days = Math.floor((today-d)/(1000*60*60*24));
        if(days < 0){ g.dateAnomalies++; } // дата статуса в будущем — ошибка ввода, не считаем в среднее
        else { g.activeDaysSum += days; g.activeDaysCount++; }
      }
    }
  });

  const list = Object.entries(byRecruiter).map(([rec,g])=>{
    const decided = g.success + g.refused; // дошли до решённого исхода: финал ИЛИ отказ
    return {
      recruiter: rec,
      total: g.total,
      active: g.active,
      decided,
      conversion: decided ? (g.success/decided*100) : null,
      avgStuckDays: g.activeDaysCount ? (g.activeDaysSum/g.activeDaysCount) : null,
      dateAnomalies: g.dateAnomalies
    };
  });

  document.getElementById("cnt-recruiters").textContent = list.length + " чел.";
  const sorted = applySorting("tbl-recruiters", list, {
    recruiter: r=>r.recruiter, total: r=>r.total, active: r=>r.active, decided: r=>r.decided,
    conversion: r=>r.conversion??-Infinity,
    avgStuckDays: r=>r.avgStuckDays??-Infinity, dateAnomalies: r=>r.dateAnomalies
  });

  tbl.innerHTML = '<thead><tr>'+
    sortableHeader("tbl-recruiters","Рекрутер","recruiter")+
    sortableHeader("tbl-recruiters","Всего кандидатов","total","num")+
    sortableHeader("tbl-recruiters","В работе сейчас","active","num")+
    sortableHeader("tbl-recruiters","Доходимость","decided","num")+
    sortableHeader("tbl-recruiters","% Доходимости","conversion","num")+
    sortableHeader("tbl-recruiters","Ср. дней без смены статуса","avgStuckDays","num")+
    sortableHeader("tbl-recruiters","Даты в будущем","dateAnomalies","num")+
    '</tr></thead><tbody>'+
    sorted.map(r=>`<tr>
      <td>${esc(r.recruiter)}</td>
      <td class="num">${r.total}</td>
      <td class="num">${r.active}</td>
      <td class="num">${r.decided}</td>
      <td class="num">${r.conversion===null ? "—" : r.conversion.toFixed(1)}</td>
      <td class="num">${r.avgStuckDays===null ? "—" : r.avgStuckDays.toFixed(0)}</td>
      <td class="num">${r.dateAnomalies>0 ? `<span class="badge warn">${r.dateAnomalies}</span>` : "0"}</td>
    </tr>`).join("") + "</tbody>";
}

function renderTransferEffectiveness(f){
  const fNoRecruiter = {...f, recruiter: ""};
  const allTransfers = collectionData.filter(r=>r.status==="Передан др. рекрутеру" && matchesBaseFilters(r, fNoRecruiter) && (!f.year || r.knStatusYear===f.year));
  const resolvedInScope = transferResolutions.filter(t=> matchesBaseFilters(t.sendRow, fNoRecruiter) && (!f.year || t.sendRow.knStatusYear===f.year));

  document.getElementById("m-transfer-total").textContent = allTransfers.length;
  document.getElementById("m-transfer-resolved").textContent = resolvedInScope.length;
  const pct = allTransfers.length ? (resolvedInScope.length/allTransfers.length*100).toFixed(1) : "0";
  document.getElementById("transfer-coverage-text").textContent = `${resolvedInScope.length} из ${allTransfers.length} (${pct}%)`;

  const outcome = (row)=>{
    if(row.status==="Трудоустроен" || row.status==="на ОП") return "success";
    if(row.status==="Отказ") return "refused";
    return "pending";
  };

  const receiverStats = {}, senderStats = {};
  resolvedInScope.forEach(t=>{
    const o = outcome(t.receivingRow);
    if(!receiverStats[t.to]) receiverStats[t.to] = {success:0, refused:0, pending:0};
    receiverStats[t.to][o]++;
    if(!senderStats[t.from]) senderStats[t.from] = {success:0, refused:0, pending:0};
    senderStats[t.from][o]++;
  });

  function renderSide(statsObj, tblId, cntId, totalLabel){
    const tbl = document.getElementById(tblId);
    const list = Object.entries(statsObj).map(([rec,s])=>{
      const total = s.success+s.refused+s.pending;
      const decided = s.success+s.refused;
      return {recruiter:rec, total, success:s.success, refused:s.refused, pending:s.pending,
        rate: decided ? (s.success/decided*100) : null};
    });
    document.getElementById(cntId).textContent = list.length + " чел.";
    if(!list.length){ tbl.innerHTML = '<tr><td class="empty-state">Нет данных с установленным исходом по текущему фильтру.</td></tr>'; return; }
    const sorted = applySorting(tblId, list, {
      recruiter:r=>r.recruiter, total:r=>r.total, success:r=>r.success, refused:r=>r.refused,
      pending:r=>r.pending, rate:r=>r.rate??-Infinity
    });
    tbl.innerHTML = '<thead><tr>'+
      sortableHeader(tblId,"Рекрутер","recruiter")+
      sortableHeader(tblId,totalLabel,"total","num")+
      sortableHeader(tblId,"Из них трудоустроены","success","num")+
      sortableHeader(tblId,"Из них отказались","refused","num")+
      sortableHeader(tblId,"Ещё в работе","pending","num")+
      sortableHeader(tblId,"% трудоустройства","rate","num")+
      '</tr></thead><tbody>'+
      sorted.map(r=>`<tr>
        <td>${esc(r.recruiter)}</td>
        <td class="num">${r.total}</td>
        <td class="num">${r.success}</td>
        <td class="num">${r.refused}</td>
        <td class="num">${r.pending}</td>
        <td class="num">${r.rate===null?"—":r.rate.toFixed(1)}</td>
      </tr>`).join("") + "</tbody>";
  }

  renderSide(receiverStats, "tbl-transfer-receiver", "cnt-transfer-receiver", "Всего получено от других рекрутеров");
  renderSide(senderStats, "tbl-transfer-sender", "cnt-transfer-sender", "Всего передано другим рекрутерам");
}

/* ==================== ДАШБОРД (воспроизводит старый дашборд руководителя) ==================== */
function renderDashboard(f){
  // Открытые заявки (с учётом фильтров, кроме Периода) — используются для связки кандидатов с текущей потребностью и для таблиц "Потребность"
  const openZayavki = reportData.filter(r=> r.status==="Открыта" && matchesBaseFilters(r, f)); // открытая заявка видна всегда, независимо от года создания
  // номера открытых заявок — для связки "Отправлено/На ОП" именно с текущей потребностью, а не со всей историей
  const openZayavkaSet = new Set(openZayavki.map(r=>r.zayavkaNum));
  const hasOpenZayavka = (r)=> r.zayavka && !/^без заявки$/i.test(r.zayavka) && openZayavkaSet.has(r.zayavka);

  // заявка -> привязанные кандидаты (используется и для колец за 2026 год, и для диаграммы "закрыли/осталось" по ОП ниже)
  const byZayavkaAll = new Map();
  collectionData.forEach(r=>{
    if(!r.zayavka || /^без заявки$/i.test(r.zayavka)) return;
    if(!byZayavkaAll.has(r.zayavka)) byZayavkaAll.set(r.zayavka, []);
    byZayavkaAll.get(r.zayavka).push(r);
  });
  const ostalosForZayavka = (r)=>{
    const linked = byZayavkaAll.get(r.zayavkaNum) || [];
    const kuplenBilet = linked.filter(c=>c.status==="Куплен билет").length;
    const naOP = linked.filter(c=>NA_OP_3.includes(c.status)).length;
    return Math.max(0, (r.kolVacancy||0) - kuplenBilet - naOP);
  };

  // Кольцо 1: заявки, поступившие в выбранном году (фильтр "Год" сверху; "Все" — весь период целиком) — сколько из них ещё открыто
  const zayavkiScope = reportData.filter(r=>{
    if(!matchesBaseFilters(r, f)) return false;
    const d = parseRUDate(r.dateReceived);
    if(!d) return false;
    if(f.year && d.getFullYear()!==f.year) return false;
    return true;
  });
  const totalZayavkiScope = zayavkiScope.length;
  const openZayavkiScopeRows = zayavkiScope.filter(r=>r.status==="Открыта");
  const openZayavkiScope = openZayavkiScopeRows.length;
  const ringZayavkiPercent = totalZayavkiScope ? (openZayavkiScope/totalZayavkiScope*100) : 0;

  // Кольцо 2: знаменатель — сумма "Кол-во вакансий" по ВСЕМ заявкам выбранного периода (любой статус).
  // Числитель ("осталось закрыть", в центре) — считаем ТОЛЬКО по заявкам, которые сейчас в статусе "Открыта":
  // закрытая заявка выбывает из остатка целиком, её кандидатов по статусам больше не пересчитываем
  // (иначе задвоение/устаревшая привязка кандидатов, которые с той заявки уже давно ушли на другую).
  const totalVacanciesScope = zayavkiScope.reduce((acc,r)=>acc+(r.kolVacancy||0), 0);
  const remainingVacanciesScope = openZayavkiScopeRows.reduce((acc,r)=>acc+ostalosForZayavka(r), 0);
  const ringVacanciesPercent = totalVacanciesScope ? (remainingVacanciesScope/totalVacanciesScope*100) : 0;

  const inScope = collectionData.filter(r=> matchesBaseFilters(r, f) && (!f.year || r.knStatusYear===f.year));
  // "В работе": группа STATUS_GROUP_1 БЕЗ статуса "на ЦТС" и БЕЗ тех, кто помечен каналом "через ЦТС" (подтверждено пользователем)
  const inWork = inScope.filter(r=> STATUS_GROUP_1.includes(r.status) && r.status!=="на ЦТС").length; // через ЦТС снова учитывается, статус "на ЦТС" по-прежнему исключён
  const cherezCts = inScope.filter(r=> STATUS_GROUP_1.includes(r.status) && r.throughCTS).length; // активные в воронке И помечены каналом

  // Объём работы за ТЕКУЩУЮ календарную неделю (Пн-Вс) — снэпшот, не зависит от фильтра "Период" сверху,
  // но учитывает остальные срезы (ОП/Проект/Вид/Категория/Рекрутер). Определения подтверждены директором:
  // "Закрыто" — кандидат получил статус на ОП / Трудоустроен / Куплен билет на этой неделе (Перевод не считаем).
  // "Открыто" — заявки, которые сейчас в статусе "Открыта" и пришли (Дата поступления заявки) на этой неделе; суммируем "Кол-во вакансий".
  const CLOSED_THIS_WEEK_STATUSES = ["на ОП","Трудоустроен","Куплен билет"];
  const nowKN = isoWeekParts(new Date());
  const closedThisWeek = collectionData.filter(r=>
    CLOSED_THIS_WEEK_STATUSES.includes(r.status) && matchesBaseFilters(r, f) &&
    r.knStatus===nowKN.week && r.knStatusYear===nowKN.year
  ).length;
  const openedThisWeek = reportData.filter(r=>{
    if(r.status!=="Открыта" || !matchesBaseFilters(r, f)) return false;
    const d = parseRUDate(r.dateReceived);
    if(!d) return false;
    const kn = isoWeekParts(d);
    return kn.week===nowKN.week && kn.year===nowKN.year;
  }).reduce((acc,r)=>acc+(r.kolVacancy||0), 0);
  document.getElementById("dash-vacancies-delta").innerHTML =
    `<span class="dyn-open">+${openedThisWeek} открыто вакансий</span>` +
    `<span class="dyn-sep">·</span>` +
    `<span class="dyn-closed">−${closedThisWeek} закрыто вакансий</span>`;
  document.getElementById("dash-week-label").textContent = `(КН ${nowKN.week})`;
  document.getElementById("dash-cherez-cts").textContent = cherezCts;
  document.getElementById("dash-inwork").textContent = inWork;
  renderRing("ring-percent", ringZayavkiPercent, "var(--blueprint)", String(openZayavkiScope));
  document.getElementById("ring-percent-total").textContent = totalZayavkiScope;
  renderRing("ring-dohodimost", ringVacanciesPercent, "var(--blueprint)", String(remainingVacanciesScope));
  document.getElementById("ring-dohodimost-total").textContent = totalVacanciesScope;

  const countStatus = (status)=> inScope.filter(r=>r.status===status).length;
  // "+N за неделю" — сколько кандидатов вошли в этот статус на ТЕКУЩЕЙ календарной неделе (knStatus = дата статуса)
  const weeklyDelta = (matchFn)=> collectionData.filter(r=> matchesBaseFilters(r, f) && matchFn(r) && r.knStatus===nowKN.week && r.knStatusYear===nowKN.year).length;
  const renderDelta = (elId, count)=>{
    const el = document.getElementById(elId);
    if(!el) return;
    if(count>0){ el.textContent = `+${count} за нед.`; el.className = "delta-lbl up"; }
    else { el.textContent = "+0 за нед."; el.className = "delta-lbl zero"; }
  };
  document.getElementById("dash-soglasovanie").textContent = countStatus("Согласование");
  document.getElementById("dash-sbordok").textContent = countStatus("Сбор документов");
  document.getElementById("dash-sb").textContent = countStatus("Служба безопасности");
  document.getElementById("dash-medosmotr").textContent = countStatus("Медосмотр");
  document.getElementById("dash-offer").textContent = countStatus("Оффер");
  document.getElementById("dash-mobilizatsiya").textContent = countStatus("Мобилизация");
  document.getElementById("dash-cts").textContent = countStatus("на ЦТС");
  document.getElementById("dash-kuplen-bilet").textContent = countStatus("Куплен билет");
  document.getElementById("dash-na-op").textContent = countStatus("на ОП");
  document.getElementById("dash-trudoustroen").textContent = countStatus("Трудоустроен");
  document.getElementById("dash-rezerv").textContent = countStatus("Резерв");
  document.getElementById("dash-otkaz").textContent = countStatus("Отказ");

  renderDelta("delta-soglasovanie", weeklyDelta(r=>r.status==="Согласование"));
  renderDelta("delta-sbordok", weeklyDelta(r=>r.status==="Сбор документов"));
  renderDelta("delta-sb", weeklyDelta(r=>r.status==="Служба безопасности"));
  renderDelta("delta-medosmotr", weeklyDelta(r=>r.status==="Медосмотр"));
  renderDelta("delta-offer", weeklyDelta(r=>r.status==="Оффер"));
  renderDelta("delta-mobilizatsiya", weeklyDelta(r=>r.status==="Мобилизация"));
  renderDelta("delta-cherez-cts", weeklyDelta(r=> STATUS_GROUP_1.includes(r.status) && r.throughCTS));
  renderDelta("delta-kuplen-bilet", weeklyDelta(r=>r.status==="Куплен билет"));
  renderDelta("delta-cts", weeklyDelta(r=>r.status==="на ЦТС"));
  renderDelta("delta-na-op", weeklyDelta(r=>r.status==="на ОП"));
  renderDelta("delta-trudoustroen", weeklyDelta(r=>r.status==="Трудоустроен"));
  renderDelta("delta-rezerv", weeklyDelta(r=>r.status==="Резерв"));
  renderDelta("delta-otkaz", weeklyDelta(r=>r.status==="Отказ"));

  // Потребность по заявкам: строки - категория, столбцы - ОП, значение - сумма Кол-во вакансий (только открытые заявки)
  const needRows = openZayavki.filter(r=>r.category);
  const needTbl = document.getElementById("tbl-dash-need");
  document.getElementById("cnt-dash-need").textContent = needRows.length + " заявок";
  if(!needRows.length){
    needTbl.innerHTML = '<tr><td class="empty-state">Нет данных по открытым заявкам с указанной категорией.</td></tr>';
    document.getElementById("chart-dash-need-op").innerHTML = '<div class="empty-state">Нет данных для диаграммы.</div>';
    document.getElementById("tbl-dash-cat").innerHTML = '<tr><td class="empty-state">Нет данных.</td></tr>';
    document.getElementById("cnt-dash-cat").textContent = "";
    return;
  }
  const categories = [...new Set(needRows.map(r=>r.category))].sort();
  const ops = [...new Set(needRows.map(r=>r.op).filter(Boolean))].sort();
  const grid = {};
  categories.forEach(c=>{ grid[c] = {}; ops.forEach(o=>grid[c][o]=0); });
  needRows.forEach(r=>{
    if(!r.op) return;
    grid[r.category][r.op] = (grid[r.category][r.op]||0) + (r.kolVacancy||0);
  });
  let needHtml = '<thead><tr><th>Категория</th>' + ops.map(o=>`<th class="num">${esc(o)}</th>`).join("") + '<th class="num">Итого</th></tr></thead><tbody>';
  const opTotals = {}; ops.forEach(o=>opTotals[o]=0);
  categories.forEach(c=>{
    let rowTotal = 0;
    const cells = ops.map(o=>{ const v = grid[c][o]||0; rowTotal+=v; opTotals[o]+=v; return `<td class="num">${v||""}</td>`; }).join("");
    needHtml += `<tr><td>${esc(c)}</td>${cells}<td class="num"><b>${rowTotal}</b></td></tr>`;
  });
  const grandTotal = Object.values(opTotals).reduce((a,b)=>a+b,0);
  needHtml += `<tr style="background:var(--blueprint-soft); font-weight:700;"><td>Итого</td>` + ops.map(o=>`<td class="num">${opTotals[o]}</td>`).join("") + `<td class="num">${grandTotal}</td></tr>`;
  needHtml += "</tbody>";
  needTbl.innerHTML = needHtml;

  // "Закрыли" для диаграммы: Трудоустроен / Перевод / на ОП / Куплен билет — все, кто уже фактически отправлен (по кандидатам, привязанным к этой же открытой заявке)
  const CLOSED_STATUSES = ["Трудоустроен","Перевод","на ОП","Куплен билет"];
  const closedByOp = {}; ops.forEach(o=>closedByOp[o]=0);
  needRows.forEach(r=>{
    if(!r.op) return;
    const linked = byZayavkaAll.get(r.zayavkaNum) || [];
    closedByOp[r.op] += linked.filter(c=>CLOSED_STATUSES.includes(c.status)).length;
  });

  const chartEntries = ops.map(o=>({label:o, total:opTotals[o], closed:Math.min(closedByOp[o], opTotals[o])})).sort((a,b)=>b.total-a.total);
  renderStackedColumnChart(document.getElementById("chart-dash-need-op"), chartEntries);

  // Потребность по категориям: Вакансий / На ОП / Осталось / В работе
  const catTbl = document.getElementById("tbl-dash-cat");
  const catOrder = ["ИТР","ИТР-Лин.","РОП","РОП-Сварщик"].filter(c=>categories.includes(c));
  const catFinal = catOrder.length ? catOrder : categories;
  document.getElementById("cnt-dash-cat").textContent = catFinal.length + " категорий";

  const catStats = catFinal.map(c=>{
    const vac = needRows.filter(r=>r.category===c).reduce((a,r)=>a+(r.kolVacancy||0),0);
    const naOP = inScope.filter(r=>r.category===c && STATUS_GROUP_2.includes(r.status) && hasOpenZayavka(r)).length;
    const ostalos = vac - naOP;
    const vRabote = inScope.filter(r=>r.category===c && STATUS_GROUP_1.includes(r.status)).length;
    return {category:c, vac, naOP, ostalos, vRabote};
  });

  catTbl.innerHTML = '<thead><tr><th>Категория</th><th class="num">Вакансий</th><th class="num">На ОП</th><th class="num">Осталось</th><th class="num">В работе</th></tr></thead><tbody>' +
    catStats.map(s=>`<tr>
      <td>${esc(s.category)}</td>
      <td class="num">${s.vac}</td>
      <td class="num">${s.naOP}</td>
      <td class="num">${s.ostalos}</td>
      <td class="num">${s.vRabote}</td>
    </tr>`).join("") + "</tbody>";
}
